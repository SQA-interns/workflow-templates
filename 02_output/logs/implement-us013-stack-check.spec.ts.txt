import AxeBuilder from "@axe-core/playwright";
import { execFileSync } from "node:child_process";
import { expect, test, type Page } from "@playwright/test";
import {
  ANA, MARKO, PETRA, api, cancelDialog, confirmDialog, createContact, createGroup, editContact,
  moveContactToRecycleBin, resetStack, signIn, sql, user,
} from "../../acceptance/support/stack";

test.beforeEach(resetStack);

/** "Konferenca 2025": Ana Owner, Marko Editor, Petra Viewer; memberships by SQL until US-004. */
async function team() {
  const ana = await user(ANA, "Ana Kovač");
  const marko = await user(MARKO, "Marko Kranjc");
  const petra = await user(PETRA, "Petra Horvat");
  const groupId = await createGroup(ana, "Konferenca 2025");
  for (const [u, role] of [[marko, "EDITOR"], [petra, "VIEWER"]] as const) {
    const id = (await api(u, "GET", "/me")).json.id;
    sql(`INSERT INTO membership VALUES ('${groupId}', '${id}', '${role}')`);
  }
  const anaId = (await api(ana, "GET", "/me")).json.id;
  const markoId = (await api(marko, "GET", "/me")).json.id;
  return { ana, marko, petra, groupId, anaId, markoId };
}
type Team = Awaited<ReturnType<typeof team>>;

const log = (k: string, v: unknown) => console.log(k, JSON.stringify(v));
const path = (t: Team, id: string) => `/groups/${t.groupId}/contacts/${id}`;
const del = (u: Team["ana"], t: Team, id: string) =>
  api(u, "DELETE", `/groups/${t.groupId}/recycle-bin/${id}`);
const bin = async (t: Team) => (await api(t.ana, "GET", `/groups/${t.groupId}/recycle-bin`)).json;
const psql = (q: string) =>
  execFileSync("docker", ["exec", "kontakti-postgres-1", "sh", "-c", `psql -U "$POSTGRES_USER" -d "$POSTGRES_DB" -tAc "${q}"`])
    .toString()
    .trim();
const rows = (table: string, id: string) => psql(`SELECT count(*) FROM ${table} WHERE contact_id = '${id}'`);
const audit = (t: Team) => psql(`SELECT user_id || ' ' || action || ' ' || target_id FROM audit_entry WHERE group_id = '${t.groupId}' ORDER BY at`);

async function janezInRecycleBin(t: Team) {
  const id = await createContact(t.ana, t.groupId, {
    fields: { firstName: "Janez", lastName: "Novak", email: "janez.novak@example.si" },
    tags: ["udelezenec2025"],
    attributes: [{ key: "kompetenca", value: "spletnoProgramiranje" }],
  });
  await editContact(t.marko, t.groupId, id, (u) => (u.fields.city = "Ljubljana"));
  await editContact(t.marko, t.groupId, id, (u) => (u.fields.phone = "+386 41 123 456"));
  await moveContactToRecycleBin(t.marko, t.groupId, id);
  return id;
}

test("API AC-013-01..03, EX-013-02, EX-013-04", async () => {
  const t = await team();
  const id = await janezInRecycleBin(t);
  // AC-013-02
  let r = await del(t.marko, t, id);
  log("AC-013-02", [r.status, r.json.code, (await bin(t)).length]);
  expect([r.status, r.json.code]).toEqual([403, "auth.forbidden"]);
  // AC-013-03
  const maja = await createContact(t.ana, t.groupId, { fields: { firstName: "Maja", lastName: "Zupan" } });
  r = await del(t.ana, t, maja);
  log("AC-013-03", [r.status, r.json.code, (await api(t.ana, "GET", path(t, maja))).json.versionNumber]);
  expect([r.status, r.json.code]).toEqual([409, "contact.notInRecycleBin"]);
  // AC-013-01
  r = await del(t.ana, t, id);
  const entries = audit(t);
  log("AC-013-01", [
    r.status, (await bin(t)).length, (await api(t.ana, "GET", path(t, id))).status,
    (await api(t.ana, "GET", `${path(t, id)}/versions`)).status,
    ["contact_version", "attribute", "contact_tag"].map((x) => rows(x, id)),
    psql(`SELECT count(*) FROM contact WHERE id = '${id}'`), entries,
  ]);
  expect(r.status).toBe(204);
  expect(entries).toBe(`${t.anaId} contact.permanentlyDeleted ${id}`);
  expect(entries).not.toMatch(/Janez|Novak|janez\.novak/);
  // EX-013-02
  const tags = (await api(t.marko, "GET", `/groups/${t.groupId}/tags?q=udel`)).json;
  log("EX-013-02", tags);
  expect(tags).toEqual([]);
  // EX-013-04
  r = await del(t.ana, t, "00000000-0000-4000-8000-000000000999");
  log("EX-013-04", [r.status]);
  expect(r.status).toBe(404);
});

test("API BR-08 removal writes contact.removed by the User who moved it (job at backend start)", async () => {
  const t = await team();
  const id = await janezInRecycleBin(t);
  psql(`UPDATE contact SET deleted_at = now() - interval '30 days 1 hour' WHERE id = '${id}'`);
  execFileSync("docker", ["restart", "kontakti-backend-1"]);
  for (let i = 0; i < 60; i++) {
    const s = execFileSync("docker", ["inspect", "-f", "{{.State.Health.Status}}", "kontakti-backend-1"]).toString().trim();
    if (s === "healthy") break;
    await new Promise((res) => setTimeout(res, 2000));
  }
  const entries = audit(t);
  log("BR-08", [(await api(t.ana, "GET", path(t, id))).status, entries]);
  expect(entries).toBe(`${t.markoId} contact.removed ${id}`);
});

async function shot(page: Page, dir: string, name: string) {
  for (const [w, h, label] of [[390, 844, "mobile"], [820, 1180, "tablet"], [1440, 900, "desktop"]] as const) {
    await page.setViewportSize({ width: w, height: h });
    await page.waitForTimeout(300);
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth > window.innerWidth);
    expect(overflow, `${name} ${label} horizontal overflow`).toBe(false);
    await page.screenshot({ path: `../logs/screens/${dir}/${name}_${label}.png`, fullPage: true });
  }
  const axe = await new AxeBuilder({ page }).analyze();
  const bad = axe.violations.filter((v) => v.impact === "critical" || v.impact === "serious");
  console.log(`axe ${dir} ${name}:`, bad.map((v) => `${v.impact} ${v.id}`).join(", ") || "none");
  expect(bad).toEqual([]);
  await page.setViewportSize({ width: 1280, height: 720 });
}

const DIR = "WF-11_recycle-bin/permanent-delete";

test("UI AC-013-01 (frozen steps) with screens", async ({ page }) => {
  const t = await team();
  const id = await createContact(t.ana, t.groupId, { fields: { firstName: "Janez", lastName: "Novak" } });
  await moveContactToRecycleBin(t.marko, t.groupId, id);
  await signIn(page, t.ana);
  await page.goto(`/groups/${t.groupId}/recycle-bin`);
  const row = page.getByRole("row", { name: /Janez Novak/ });
  await expect(row.getByRole("button", { name: "Delete permanently" })).toBeVisible();
  await shot(page, DIR, "owner");
  await row.getByRole("button", { name: "Delete permanently" }).click();
  await expect(page.getByRole("alertdialog")).toContainText("Delete Janez Novak permanently?");
  await shot(page, DIR, "confirmation");
  await confirmDialog(page, "Delete permanently");
  await expect(page.getByText("The recycle bin is empty.")).toBeVisible();
  expect((await api(t.ana, "GET", path(t, id))).status).toBe(404);
  await page.goto(`/groups/${t.groupId}/contacts/${id}`);
  await expect(page.getByText("This page cannot be found.")).toBeVisible();
  console.log("UI AC-013-01 passed");
});

test("UI EX-013-01 (frozen steps); Editor without the action", async ({ page, browser }) => {
  const t = await team();
  const id = await createContact(t.ana, t.groupId, { fields: { firstName: "Janez", lastName: "Novak" } });
  await moveContactToRecycleBin(t.marko, t.groupId, id);
  await signIn(page, t.ana);
  await page.goto(`/groups/${t.groupId}/recycle-bin`);
  await page.getByRole("row", { name: /Janez Novak/ }).getByRole("button", { name: "Delete permanently" }).click();
  await cancelDialog(page);
  await expect(page.getByRole("row", { name: /Janez Novak/ })).toBeVisible();
  const entries = await bin(t);
  expect(entries.map((e: { contact: { name: string } }) => e.contact.name)).toEqual(["Janez Novak"]);
  console.log("UI EX-013-01 passed");
  const editor = await (await browser.newContext()).newPage();
  await signIn(editor, t.marko);
  await editor.goto(`/groups/${t.groupId}/recycle-bin`);
  await expect(editor.getByRole("button", { name: "Restore" })).toBeVisible();
  await expect(editor.getByRole("button", { name: "Delete permanently" })).toHaveCount(0);
  await shot(editor, DIR, "editor");
  console.log("UI Editor: Restore only");
});
