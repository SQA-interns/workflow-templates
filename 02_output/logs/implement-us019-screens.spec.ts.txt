import AxeBuilder from "@axe-core/playwright";
import { expect, test, type Page } from "@playwright/test";
import { api, createContact, editContact, resetStack, signIn, team } from "./support/stack";

// Throwaway check of WF-07 (Undo merge) for US-019 (build-ui steps 4 and 5); not committed.
test.beforeEach(resetStack);

async function shot(page: Page, screen: string, name: string) {
  for (const [w, h, label] of [[390, 844, "mobile"], [820, 1180, "tablet"], [1440, 900, "desktop"]] as const) {
    await page.setViewportSize({ width: w, height: h });
    await page.waitForTimeout(300);
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth > window.innerWidth);
    if (overflow) {
      await page.screenshot({ path: `../logs/screens/${screen}/${name}_${label}_overflow.png`, fullPage: true });
      console.log("wide:", await page.evaluate(() => [...document.querySelectorAll("body *")].filter((e) => e.getBoundingClientRect().right > window.innerWidth + 1).slice(0, 8).map((e) => `${e.tagName}.${(e as HTMLElement).className} w=${Math.round(e.getBoundingClientRect().width)}`)));
    }
    expect(overflow, `${screen} ${name} ${label} horizontal overflow`).toBe(false);
    await page.screenshot({ path: `../logs/screens/${screen}/${name}_${label}.png`, fullPage: true });
  }
  const axe = await new AxeBuilder({ page }).analyze();
  const bad = axe.violations.filter((v) => v.impact === "critical" || v.impact === "serious");
  console.log(`axe ${screen} ${name}:`, bad.map((v) => `${v.impact} ${v.id}`).join(", ") || "none");
  expect(bad).toEqual([]);
  await page.setViewportSize({ width: 1280, height: 720 });
}

test("US-019 screens", async ({ page }) => {
  const t = await team();
  const janez = await createContact(t.ana, t.groupId, {
    fields: { firstName: "Janez", lastName: "Novak", email: "janez.novak@example.si" },
    tags: ["udelezenec2025"],
  });
  const other = await createContact(t.ana, t.groupId, {
    fields: { firstName: "J.", lastName: "Novak", email: "janez.novak@example.si", phone: "+386 41 123 456" },
    tags: ["vip"],
  });
  const res = await api(t.marko, "POST", `/groups/${t.groupId}/merges`, {
    primaryId: janez, otherId: other, primaryVersionNumber: 1, otherVersionNumber: 1,
    fieldChoices: { phone: "other" },
  });
  expect(res.status).toBe(201);
  await signIn(page, t.marko);
  const details = `/groups/${t.groupId}/contacts/${janez}`;

  await page.goto(details);
  await expect(page.getByRole("button", { name: "Undo merge" })).toBeVisible();
  await shot(page, "WF-07_contact-details", "undo-merge");

  await page.getByRole("button", { name: "Undo merge" }).click();
  await expect(page.getByRole("alertdialog", { name: "Undo merge?" })).toBeVisible();
  await shot(page, "WF-07_contact-details", "undo-merge-confirm");
  await page.getByRole("button", { name: "Cancel" }).click();

  await editContact(t.ana, t.groupId, janez, (u) => (u.fields.jobTitle = "CEO"));
  await page.goto(details);
  await page.getByRole("button", { name: "Undo merge" }).click();
  await page.getByRole("alertdialog").getByRole("button", { name: "Undo merge" }).click();
  await expect(page.getByText("The Contact changed after the Merge. Restore the Versions you need instead.")).toBeVisible();
  await shot(page, "WF-07_contact-details", "undo-merge-refused");
});
